        { icon: '🛂', title: 'EMBARQUE', texto: `Basta mostrar o PDF do bilhete ao motorista pela tela do celular junto com o documento original físico com foto, mas caso deseje imprimir os bilhetes, basta se dirigir até o guichê e imprimir o bilhete.` },

    let observer = null;

    function garantirEstilo() {
        if (document.getElementById(ESTILO_ID)) return;
        const style = document.createElement('style');
        style.id = ESTILO_ID;
        style.textContent = `
            footer.chat-messages-footer.tm-com-barra { flex-wrap: wrap !important; }
            #${GRUPO_ID} {
                flex: 0 0 100%; order: -1; width: 100%;
                display: flex; flex-wrap: wrap; gap: 3px;
                padding: 0 0 6px 0; margin: 0;
            }
            #${GRUPO_ID} button { position: relative; }
            #${GRUPO_ID} button::after {
                content: attr(data-tooltip);
                position: absolute; top: -34px; left: 50%;
                transform: translateX(-50%);
                background: #000; color: #fff;
                padding: 4px 8px; border-radius: 6px;
                font-size: 11px; white-space: nowrap;
                opacity: 0; pointer-events: none;
                transition: opacity .15s ease; z-index: 9999;
            }
            #${GRUPO_ID} button:hover::after { opacity: 1; }
        `;
        document.head.appendChild(style);
    }

    function noChat() {
        // ChatPro: /chat/<id>  |  Tallos: /app/chat
        return /\/chat(\/|$)/.test(location.pathname);
    }

    function esconderSugestaoResposta() {
        document.querySelectorAll(SUGESTAO_SELECTOR).forEach(btn => { btn.style.display = 'none'; });
    }

    function escaparHtml(s) {
        return s.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;');
    }

    function acharEditor(footer) {
        for (const sel of EDITOR_SELECTORS) {
            const el = footer.querySelector(sel);
            if (el && !el.closest('#' + GRUPO_ID)) return el;
        }
        return null;
    }

    function inserirTexto(texto, btn) {
        const footer = document.querySelector(FOOTER_SELECTOR);
        const editor = footer && acharEditor(footer);
        if (!editor) return;

        texto = texto.replace(/\r/g, '');

        if (editor.tagName === 'TEXTAREA' || editor.tagName === 'INPUT') {
            // campo nativo (React/Vue): usa o setter do protótipo para o framework perceber
            const proto = editor.tagName === 'TEXTAREA' ? HTMLTextAreaElement.prototype : HTMLInputElement.prototype;
            Object.getOwnPropertyDescriptor(proto, 'value').set.call(editor, texto);
            editor.dispatchEvent(new Event('input', { bubbles: true }));
            editor.dispatchEvent(new Event('change', { bubbles: true }));
        } else if (editor.classList.contains('ql-editor')) {
            // editor Quill
            editor.innerHTML = texto
                .split('\n')
                .map(l => l.trim() === '' ? '<p><br></p>' : `<p>${escaparHtml(l)}</p>`)
                .join('');
            editor.dispatchEvent(new Event('input', { bubbles: true }));
        } else {
            // contenteditable genérico: substitui o conteúdo como se fosse digitado
            editor.focus();
            const sel = window.getSelection();
            const range = document.createRange();
            range.selectNodeContents(editor);
            sel.removeAllRanges();
            sel.addRange(range);
            if (!document.execCommand('insertText', false, texto)) {
                editor.innerText = texto;
                editor.dispatchEvent(new Event('input', { bubbles: true }));
            }
        }

        editor.focus();

        const original = btn.innerHTML;
        btn.innerHTML = '✅';
        setTimeout(() => { btn.innerHTML = original; }, 800);
    }

    function inserirBotoes() {
        const footer = document.querySelector(FOOTER_SELECTOR);
        if (!footer || footer.querySelector('#' + GRUPO_ID)) return;

        const grupo = document.createElement('div');
        grupo.id = GRUPO_ID;

        MENSAGENS.forEach(msg => {
            const btn = document.createElement('button');
            btn.type = 'button';
            btn.innerHTML = msg.icon.trim();
            btn.setAttribute('data-tooltip', msg.title);
            btn.style.cssText = `
                width:28px;height:28px;background:transparent;border:none;
                border-radius:6px;font-size:16px;cursor:pointer;opacity:.75;
                display:flex;align-items:center;justify-content:center;padding:0;
            `;
            btn.onmouseenter = () => { btn.style.opacity = '1'; };
            btn.onmouseleave = () => { btn.style.opacity = '.75'; };
            // mousedown + preventDefault: o clique não tira o foco do campo de mensagem
            btn.onmousedown = e => e.preventDefault();
            btn.onclick = () => inserirTexto(msg.texto, btn);
            grupo.appendChild(btn);
        });

        // entra como primeiro filho do footer, ocupando a linha toda acima do campo
        footer.classList.add('tm-com-barra');
        footer.insertBefore(grupo, footer.firstChild);
    }

    function rodar() {
        if (!noChat()) return;          // site SPA: confere a rota a cada mudança
        esconderSugestaoResposta();
        inserirBotoes();
    }

    return {
        id: 'chat_barras_mensagens',
        name: 'CHAT - Barras e mensagens',
        defaultOn: true,

        init() {
            garantirEstilo();
            observer = new MutationObserver(rodar);
            observer.observe(document.body, { childList: true, subtree: true });
            rodar();
        },

        destroy() {
            observer?.disconnect();
            observer = null;
            document.getElementById(GRUPO_ID)?.remove();
            document.getElementById(ESTILO_ID)?.remove();
            document.querySelectorAll('footer.tm-com-barra').forEach(f => f.classList.remove('tm-com-barra'));
            document.querySelectorAll(SUGESTAO_SELECTOR).forEach(btn => { btn.style.display = ''; });
        }
    };
}

registrarModulo(criarModuloBarrasMensagens());
